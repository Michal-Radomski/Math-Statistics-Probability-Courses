Here is how you can calculate the **mean**, **median**, and **sample standard deviation** using JavaScript or TypeScript.

```typescript
// 1. Calculate the Mean
function calculateMean(arr: number[]): number {
  if (arr.length === 0) return 0;
  const sum = arr.reduce((acc, val) => acc + val, 0);
  return sum / arr.length;
}

// 2. Calculate the Median
function calculateMedian(arr: number[]): number {
  if (arr.length === 0) return 0;

  // Create a sorted copy so we don't mutate the original array
  const sorted = [...arr].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);

  // If even number of elements, average the two middle values
  if (sorted.length % 2 === 0) {
    return (sorted[mid - 1] + sorted[mid]) / 2;
  }

  // If odd number of elements, return the exact middle
  return sorted[mid];
}

// 3. Calculate the Sample Standard Deviation
function calculateStandardDeviation(arr: number[]): number {
  if (arr.length <= 1) return 0;

  const mean = calculateMean(arr);

  // Sum the squared differences from the mean
  const squaredDiffsSum = arr.reduce((acc, val) => {
    const diff = val - mean;
    return acc + diff * diff;
  }, 0);

  // Divide by n - 1 for sample standard deviation, then take the square root
  const variance = squaredDiffsSum / (arr.length - 1);
  return Math.sqrt(variance);
}

// --- Example Usage ---
const data = [2, 4, 5, 6, 8];

console.log("Mean:", calculateMean(data)); // 5
console.log("Median:", calculateMedian(data)); // 5
console.log("Standard Deviation:", calculateStandardDeviation(data)); // ~2.236
```
